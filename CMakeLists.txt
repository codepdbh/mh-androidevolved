cmake_minimum_required(VERSION 3.22)
project(manhunt_runtime LANGUAGES C)

add_library(manhunt_guest_memory STATIC runtime/guest_memory.c)
target_include_directories(manhunt_guest_memory PUBLIC runtime)
target_compile_features(manhunt_guest_memory PUBLIC c_std_11)
add_library(manhunt_pe32 STATIC runtime/pe32_image.c runtime/pe32_imports.c runtime/pe32_import_plan.c)
target_link_libraries(manhunt_pe32 PUBLIC manhunt_guest_memory)
target_compile_features(manhunt_pe32 PUBLIC c_std_11)

include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
    add_executable(test_guest_memory tests/test_guest_memory.c)
    target_link_libraries(test_guest_memory PRIVATE manhunt_guest_memory)
    add_test(NAME guest_memory COMMAND test_guest_memory)
    add_executable(test_pe_image tests/test_pe_image.c)
    target_link_libraries(test_pe_image PRIVATE manhunt_pe32)
    add_test(NAME pe_image COMMAND test_pe_image)
    add_executable(test_checked_rep_movs tests/test_checked_rep_movs.c)
    target_include_directories(test_checked_rep_movs PRIVATE recomp)
    target_link_libraries(test_checked_rep_movs PRIVATE manhunt_guest_memory)
    add_test(NAME checked_rep_movs COMMAND test_checked_rep_movs)
endif()
